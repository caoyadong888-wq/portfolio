# Not executed here. This box has no Docker. Images are pinned by digest.
# Named volumes hold Prometheus, Alertmanager, and Grafana state. There is no
# repo directory named data. Grafana's admin password comes from the environment.
# PROMETHEUS_URL is what grafana/provisioning/datasources/prometheus.yml reads.

services:
  prometheus:
    image: prom/prometheus:v3.15.0@sha256:efd719c99d83b060d9daefdcf00360461adf279f45ef5391f8d111892118753e
    command:
      - --config.file=/etc/observastack/prometheus/prometheus.yml
      - --storage.tsdb.path=/prometheus
      - --web.listen-address=:9090
    volumes:
      - .:/etc/observastack:ro
      - prom_tsdb:/prometheus
    ports:
      - "127.0.0.1:9090:9090"
    mem_limit: 512m
    cpus: 0.50
    healthcheck:
      test: ["CMD", "wget", "-q", "-O", "/dev/null", "http://127.0.0.1:9090/-/healthy"]
      interval: 15s
      timeout: 5s
      retries: 5

  alertmanager:
    image: prom/alertmanager:v0.34.1@sha256:e9733bafb1bdef9b00e25a21f8f99dc26a22224bf16641ad754d1649f4c3357a
    command:
      - --config.file=/etc/observastack/alertmanager/alertmanager.yml
      - --storage.path=/alertmanager
      - --web.listen-address=:9093
    working_dir: /etc/observastack
    volumes:
      - .:/etc/observastack:ro

      - "127.0.0.1:9093:9093"
    mem_limit: 256m

      test: ["CMD", "wget", "-q", "-O", "/dev/null", "http://127.0.0.1:9093/-/healthy"]

      GF_PLUGINS_PREINSTALL_DISABLED: "true"

      - "127.0.0.1:3000:3000"
    mem_limit: 512m
    cpus: 0.50
    healthcheck:
      test: ["CMD", "wget", "-q", "-O", "/dev/null", "http://127.0.0.1:3000/api/health"]
      interval: 15s
      timeout: 5s
      retries: 8
