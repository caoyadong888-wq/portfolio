policies/.pint.hcl

parser {
  include = ["(.*/)?rules/.*\\.yml"]
  schema  = "prometheus"
}

prometheus "larkspur" {
  uri      = "http://127.0.0.1:19090"
  timeout  = "5s"
  required = false
}

rule {
  match {
    kind = "alerting"
  }

  annotation "summary" {
    required = true
    severity = "bug"
  }

  annotation "description" {
    required = true
    severity = "bug"
  }

  annotation "runbook_url" {
    required = true
    severity = "bug"
    value    = "https://runbooks\\.larkspur\\.example/[A-Za-z0-9]+"
  }

  label "severity" {
    required = true
    severity = "bug"
    value    = "critical|warning|info"
  }

  label "team" {
    required = true
    severity = "bug"
    value    = "orders|payments|platform"
  }
}
