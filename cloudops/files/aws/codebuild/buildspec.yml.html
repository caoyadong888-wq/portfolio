# SECONDARY — AWS parallel; GCP is the primary narrative of this demo.
# 次要. The portfolio build does not start this CodeBuild project.
version: 0.2
phases:
  install:
    runtime-versions:
      python: 3.12
    commands:
      - python -m pip install --upgrade pip
      - python -m pip install -e app
  pre_build:
    commands:
      - python -m ruff check app aws/lambda
      - python -m pytest -q app/tests
  build:
    commands:
      - python -m compileall aws/lambda
  post_build:
    commands:
      - echo "SECONDARY package ready. This buildspec does not call AWS during make check."
artifacts:
  files:
    - aws/lambda/**/*
    - aws/template.yaml
