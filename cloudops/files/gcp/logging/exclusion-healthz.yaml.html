exclusion:
  name: insights-api-healthz-noise
  project: thistle-analytics-demo
  description: Drop successful probe lines so error-log alerts are not diluted by health checks.
  filter: |
    resource.type="cloud_run_revision"
    resource.labels.service_name="insights-api"
    httpRequest.requestUrl=~"/healthz"
    httpRequest.status<400
  disabled: false
