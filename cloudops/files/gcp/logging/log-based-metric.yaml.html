logMetric:
  name: insights_api_app_errors
  project: thistle-analytics-demo
  description: Application error lines from the insights-api Cloud Run revision.
  filter: |
    resource.type="cloud_run_revision"
    resource.labels.service_name="insights-api"
    severity>=ERROR
  metricDescriptor:
    metricKind: DELTA
    valueType: INT64
    unit: "1"
  labelExtractors: {}
