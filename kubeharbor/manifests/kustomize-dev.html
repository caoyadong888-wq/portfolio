  jobTemplate:
    spec:
      template:
        metadata:
          labels:
            app.kubernetes.io/component: backup

            app.kubernetes.io/name: ledgerlite
            app.kubernetes.io/part-of: ledgerlite

          automountServiceAccountToken: false
          containers:
          - command:
            - /bin/sh
            - -c
            - |

              mkdir -p /backup
              pg_dump --file="/backup/ledgerlite-${ts}.sql"
